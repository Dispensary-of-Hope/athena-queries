-- part of BundleB2B orders and products pipeline
-- creates a table of products fromBigcommerce and quantity ordered
-- can be linked to the b3 orders table

CREATE EXTERNAL TABLE products (
  `row_id` string,
  `id` string,
  `order_id` string,
  `product_id` string,
  `variant_id` string,
  `name` string,
  `sku` string,
  `upc` string,
  `type` string,
  `order_source` string,
  `quantity` bigint,
  `quantity_shipped` bigint,
  )
ROW FORMAT SERDE 'org.apache.hadoop.hive.serde2.lazy.LazySimpleSerDe'
WITH SERDEPROPERTIES (
  'serialization.format' = ',',
  'field.delim' = ','
) LOCATION 's3://b3-order-data-csv/sandbox/products/'
TBLPROPERTIES ('has_encrypted_data'='false',
              'skip.header.line.count'='1');
