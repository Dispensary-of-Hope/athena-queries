-- part of BundleB2B orders and products pipeline
-- creates a table of orders with company info from BigCommerce export to S3

CREATE EXTERNAL TABLE orders (
  `row_id` string,
  `id` string,
  `customer_id` string,
  `status_id` string,
  `status` string,
  `items_total` bigint,
  `items_shipped` bigint,
  `ip_address` string,
  `staff_notes` string,
  `order_source` string,
  `company_id` string,
  `company_name` string,
  `date_created` timestamp,
  `date_modified` timestamp,
  `date_shipped` timestamp 
)
ROW FORMAT SERDE 'org.apache.hadoop.hive.serde2.lazy.LazySimpleSerDe'
WITH SERDEPROPERTIES (
  'serialization.format' = ',',
  'field.delim' = ','
) LOCATION 's3://b3-order-data-csv/sandbox/orders/'
TBLPROPERTIES ('has_encrypted_data'='false',
              'skip.header.line.count'='1');
