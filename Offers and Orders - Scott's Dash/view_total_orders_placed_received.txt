CREATE OR REPLACE VIEW "view_total_orders_placed_received" AS(SELECT pro.year, pro.month, pro.excess_orders_placed, pro.planned_orders_placed, (pro.excess_orders_placed + pro.planned_orders_placed) total_orders_placed, total_orders_received        FROM (SELECT month(purchase_order_date) month,    year(purchase_order_date) year, count(distinct CASE WHEN order_id Like '%E%' THEN order_id END) excess_orders_placed, count(distinct CASE WHEN order_id Like '%P%' THEN order_id END) planned_orders_placed    FROM orders_historical_finale_timestamp    WHERE type = 'Purchase'    Group BY 1,2) pro    JOIN (SELECT total_orders_received, month, year FROM                (Select month(receive_date) month, year(receive_date) year, count(distinct order_id) total_orders_received                FROM orders_historical_finale_timestamp                WHERE order_id like '%E%' OR order_id like '%P%' AND type = 'Purchase'                Group BY month(receive_date), year(receive_date)                Order BY Year(receive_date) DESC, month(receive_date) DESC)) tro ON tro.month = pro.month AND tro.year = pro.yearGroup BY 1,2,3,4, 6Order BY 1 DESC, 2 DESC);
