CREATE OR REPLACE VIEW "view_manufacturer_monthly_offers" AS
(
   SELECT
    cast(date_parse(offer_date,'%m/%d/%Y') as timestamp) as offer_date 

   , "count"(DISTINCT (CASE WHEN (offer_id = '99999999-9N') THEN mfg_offerdate WHEN (offer_id <> '99999999-9N') THEN offer_id END)) offers_total
   FROM
     (
      SELECT
        offer_date
      , manufacturer
      , offer_id
      , "concat"(manufacturer, offer_date) mfg_offerdate
      FROM
        "database-prod"."manufacturer_offers"
   ) 
   GROUP BY 1
   ORDER BY 1 DESC
)
