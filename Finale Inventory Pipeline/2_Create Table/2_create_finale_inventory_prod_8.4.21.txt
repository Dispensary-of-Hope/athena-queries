-- part of Finale inventory pipeline
-- Creates table using the Finale inventory export in S3

CREATE EXTERNAL TABLE finale_inventory (ndc11 string, status string, quantity_on_hand_units bigint, available_to_promise_units bigint, lot_id string, sublocation string, purchase_order_quantity bigint)
ROW FORMAT SERDE 'org.apache.hadoop.hive.serde2.lazy.LazySimpleSerDe'WITH SERDEPROPERTIES ( 'serialization.format' = ',', 'field.delim' = ',') LOCATION 's3://finale-reports-bucket-prod/Finale Inventory/' TBLPROPERTIES ('has_encrypted_data'='false', 'skip.header.line.count'='1');
