CREATE OR REPLACE VIEW "view_top_10_distribution" AS 
SELECT
  "date_trunc"('quarter', day) day
, top_10_sites_distribution
, "sum"(total_units_distributed) total_units
, "round"((CAST(top_10_sites_distribution AS double) / "sum"(CAST(total_units_distributed AS double))), 4) top_10_percent_of_total
FROM
  (view_therapeutics_dashboard vtd2
LEFT JOIN (
   SELECT
     "date_trunc"('quarter', day) quarter
   , "sum"(total_units_distributed) top_10_sites_distribution
   FROM
     (view_therapeutics_dashboard vtd
   INNER JOIN (
      SELECT
        "date_trunc"('quarter', day)
      , "sum"(total_units_distributed)
      , site_name
      FROM
        view_therapeutics_dashboard
      WHERE ((day < "date_trunc"('quarter', current_date)) AND (day >= "date_add"('quarter', -1, "date_trunc"('quarter', current_date))))
      GROUP BY 1, 3
      ORDER BY 2 DESC
      LIMIT 10
   )  top_10 ON (top_10.site_name = vtd.site_name))
   GROUP BY 1
)  top_10 ON ("date_trunc"('quarter', vtd2.day) = top_10.quarter))
WHERE (vtd2.day >= TIMESTAMP '2019-01-01 00:00:00.000')
GROUP BY 1, 2
