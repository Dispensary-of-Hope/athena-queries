(
   SELECT
     COALESCE(total_data.receive_date, targeted_data.receive_date) receive_date
   , COALESCE(total_data.priority, targeted_data.priority) priority
   , total_data.total_targeted_units_received total_units_received
   , total_data.total_targeted_doses total_doses
   , targeted_data.total_targeted_units_received targeted_units_received
   , targeted_data.total_targeted_doses targeted_doses
   FROM
     ((
      SELECT
        receive_date
      , priority
      , SUM(CAST(total_units AS DOUBLE)) total_targeted_units_received
      , SUM((CAST(total_units AS DOUBLE) * packagesize)) total_targeted_doses
      FROM
        (orders_historical_finale_timestamp ohft
      LEFT JOIN (
         SELECT DISTINCT
           product_id
         , priority
         , productnamelong
         , donor_name
         , donor_id
         , packagesize
         FROM
           view_inventory_master
         WHERE (priority IN ('P1', 'P2', 'P3', 'P4'))
      )  inv ON ((ohft.product_id = inv.product_id) AND (inv.donor_id = ohft.donor_id)))
      WHERE (receive_date IS NOT NULL)
      GROUP BY 1, 2
   )  total_data
   FULL JOIN (
      SELECT
        receive_date
      , priority
      , SUM(CAST(total_units AS DOUBLE)) total_targeted_units_received
      , SUM((CAST(total_units AS DOUBLE) * packagesize)) total_targeted_doses
      FROM
        (orders_historical_finale_timestamp ohft
      LEFT JOIN (
         SELECT DISTINCT
           product_id
         , priority
         , productnamelong
         , donor_name
         , donor_id
         , packagesize
         FROM
           view_inventory_master
         WHERE (priority IN ('P1', 'P2', 'P3', 'P4'))
      )  inv ON ((ohft.product_id = inv.product_id) AND (inv.donor_id = ohft.donor_id)))
      WHERE ((receive_date IS NOT NULL) AND ((ohft.order_id IN ('20241015-1E', '20241014-1E', '20241021-1P', '20241105-1P', '20250108-1E', '20250110-1P', '20250128-1T', '20250220-2T', '20250313-2T', '20241101-1E', '20241206-1E', '20240121-1E', '20250523-1T', '20250505-1T', '20250516-1T', '20250121-1E')) OR (ohft.order_id LIKE '%T')))
      GROUP BY 1, 2
   )  targeted_data ON ((total_data.receive_date = targeted_data.receive_date) AND (total_data.priority = targeted_data.priority)))
   ORDER BY receive_date ASC, priority ASC
)
