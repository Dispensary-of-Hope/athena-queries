SELECT
  iq.target_list_product
, iq.priority
, ofh.day
, "sum"(ofh.total_units_received) total_units_received
, "sum"(ofh.total_in_stock) total_in_stock
, ofh.productnamelong
FROM
  (iqvia_target_primary iq
LEFT JOIN view_ordering_tool ofh ON (ofh.product_id = iq.ndc))
WHERE (target_list_product <> 'None')
GROUP BY 1, 2, 3, 6
