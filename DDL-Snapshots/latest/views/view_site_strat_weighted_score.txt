(
   SELECT DISTINCT
     st.site_name
   , ehope_type__c
   , account_manager
   , CAST(st.weighted_score AS int) weighted_score
   , (CASE WHEN (CAST(st.weighted_score AS int) < 40) THEN 'Low' WHEN ((CAST(st.weighted_score AS int) < 61) AND (CAST(st.weighted_score AS int) > 39)) THEN 'Medium' ELSE 'High' END) high_medium_low_risk
   FROM
     ("database-prod"."site_stratification" st
   INNER JOIN (
      SELECT
        site_name
      , "max"(CAST(weighted_score AS int)) weighted_score
      FROM
        site_stratification
      GROUP BY site_name
   )  st2 ON ((st.site_name = st2.site_name) AND (CAST(st.weighted_score AS int) = CAST(st2.weighted_score AS int))))
)
