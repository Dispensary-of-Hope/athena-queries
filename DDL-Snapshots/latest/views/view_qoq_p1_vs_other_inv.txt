SELECT *
FROM
  (
(
      SELECT
        quarter
      , count(DISTINCT productnamelong) unique_products
      , priority p1
      , 'false' p2_p3_p4
      , (sum(available_qty) / count(DISTINCT current_date_in_stock)) avg_p1_units_in_stock
      , 0 avg_other_units_in_stock
      FROM
        (
         SELECT
           date_trunc('quarter', current_date_in_stock) quarter
         , current_date_in_stock
         , productnamelong
         , priority
         , available_qty
         FROM
           view_running_inventory
         WHERE ((donor_id <> '702') AND (available_qty > 0) AND (priority = 'P1') AND (current_date_in_stock >= TIMESTAMP '2022-01-01 00:00:00.000'))
      ) 
      GROUP BY 1, 3, 4, 6
      ORDER BY 1 DESC
   ) UNION ALL (
      SELECT
        quarter
      , count(DISTINCT productnamelong) unique_products
      , 'false' p1
      , 'P2,P3,P4' p2_p3_p4
      , 0 avg_p1_units_in_stock
      , (sum(available_qty) / count(DISTINCT current_date_in_stock)) avg_other_units_in_stock
      FROM
        (
         SELECT
           date_trunc('quarter', current_date_in_stock) quarter
         , current_date_in_stock
         , productnamelong
         , priority
         , available_qty
         FROM
           view_running_inventory
         WHERE ((donor_id <> '702') AND (available_qty > 0) AND (priority <> 'P1') AND (current_date_in_stock >= TIMESTAMP '2022-01-01 00:00:00.000'))
      ) 
      GROUP BY 1, 3, 4, 5
      ORDER BY 1 DESC
   ) )
