(
   SELECT
     y.order_id
   , y.type
   , y.ship_date
   , y.receive_date
   , y.product_id
   , y.productnamelong
   , (CASE WHEN (y.drug_class <> '') THEN dcc2.main_drug_class ELSE y.iqvia_drugclass END) drug_class
   , y.qty
   , y.packagesize
   , (CASE WHEN ((y.iqvia_packagesize IS NOT NULL) AND (NOT (product_id IN ('72065013011', '72065013012')))) THEN "round"((y.iqvia_packagesize * y.qty), 2) WHEN ((y.iqvia_packagesize IS NULL) AND (NOT (product_id IN ('72065013011', '72065013012')))) THEN "round"((y.inv_packagesize * y.qty), 2) WHEN (product_id IN ('72065013011', '72065013012')) THEN "round"((2 * y.qty), 2) ELSE 0 END) total_doses
   , y.shippingstatecode
   , y.account_type
   , sf.donor_code__c donor_code
   , y.donor_id
   , sf.name manufacturer
   , TRY("date_parse"(sf.first_donation__c, '%m/%d/%Y')) first_donation_date
   , y.dispensary_name
   , y.shippingpostalcode
   , dt.day
   , y.iqvia_drugclass
   , y.iqvia_productnamelong
   , CAST(y.iqvia_packagesize AS varchar) iqvia_packagesize
   , y.total_wac
   , y.site_id
   , y.order_pick_day__c
   , y.status
   , 0 fills
   FROM
     (((donor_report_1 y
   INNER JOIN date_table dt ON ((dt.day = y.ship_date) OR (dt.day = y.receive_date)))
   LEFT JOIN drug_class_convert dcc2 ON (y.drug_class = dcc2.aggregated_drug_class))
   LEFT JOIN (
      SELECT DISTINCT
        name
      , donor_id__c
      , donor_code__c
      , first_donation__c
      FROM
        sf_account
      WHERE ((account_record_type_text__c IN ('Distributor', 'Manufacturers')) AND (type IN ('Active', 'Retired', 'Inactive')))
   )  sf ON (y.donor_id = sf.donor_id__c))
) UNION ALL (
   SELECT DISTINCT
     null order_id
   , 'Sale' type
   , null ship_date
   , null receive_date
   , null product_id
   , null productnamelong
   , null drug_class
   , null qty
   , null packagesize
   , 0 total_doses
   , shippingstatecode
   , null account_type
   , null donor_code
   , null donor_id
   , x.manufacturer
   , null first_donation_date
   , null dispensary_name
   , null shippingpostalcode
   , current_date day
   , null iqvia_drugclass
   , null iqvia_productnamelong
   , null iqvia_packagesize
   , 0 total_wac
   , null site_id
   , null order_pick_day__c
   , null status
   , 0 fills
   FROM
     ((
      SELECT DISTINCT name manufacturer
      FROM
        (
         SELECT DISTINCT name
         FROM
           sf_account
         WHERE ((account_record_type_text__c IN ('Distributor', 'Manufacturers')) AND (type IN ('Active', 'Retired')))
      ) 
   )  x
   CROSS JOIN sf_account)
   WHERE (shippingstatecode <> '')
) UNION ALL (
   SELECT DISTINCT
     null order_id
   , null type
   , null ship_date
   , null receive_date
   , null product_id
   , null productnamelong
   , null drug_class
   , null qty
   , null packagesize
   , 0 total_doses
   , sf.shippingstatecode
   , null account_type
   , null donor_code
   , null donor_id
   , null manufacturer
   , null first_donation_date
   , null dispensary_name
   , null shippingpostalcode
   , qn.day
   , null iqvia_drugclass
   , null iqvia_productnamelong
   , null iqvia_packagesize
   , 0 total_wac
   , null site_id
   , null order_pick_day__c
   , null status
   , qn.fills
   FROM
     (view_quarterly_numbers qn
   LEFT JOIN (
      SELECT DISTINCT
        id__c
      , shippingstatecode
      FROM
        sf_account
   )  sf ON (qn.site_id = sf.id__c))
)
