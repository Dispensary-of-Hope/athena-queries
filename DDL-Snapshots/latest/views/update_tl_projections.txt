WITH
  drd2 AS (
   SELECT
     target_list_product
   , min(day) min_ship_date
   , max(day) max_ship_date
   , sum(total_units_distributed) total_units_distributed
   , sum(total_units_received) total_units_received
   , count(DISTINCT day) number_of_days_ordered
   , round((sum(CAST(total_units_distributed AS double)) / CAST(count(DISTINCT day) AS double)), 2) average_daily_usage
   FROM
     ((distribution_receiving_destruction drd
   LEFT JOIN (
      SELECT DISTINCT
        ndc
      , target_list_product
      FROM
        iqvia_target_primary
   )  iq ON (drd.product_id = iq.ndc))
   LEFT JOIN (
      SELECT DISTINCT product_id
      FROM
        view_inventory_master
      WHERE (donor_id = '702')
   )  inv ON (drd.product_id = inv.product_id))
   WHERE ((day >= date_add('year', -1, current_date)) AND (target_list_product IS NOT NULL) AND (target_list_product <> 'None') AND ((inv.product_id IS NULL) OR (inv.product_id = '')))
   GROUP BY 1
   ORDER BY 1 ASC
) 
, vri2 AS (
   SELECT DISTINCT
     target_list_product
   , sum((CASE WHEN (current_date_in_stock = date_add('year', -1, current_date)) THEN total_in_stock END)) begin_date_stock_count
   , sum((CASE WHEN (current_date_in_stock = date_add('year', -1, current_date)) THEN total_in_stock END)) end_date_stock_count
   , count(DISTINCT current_date_in_stock) number_of_days_in_stock
   FROM
     (
      SELECT DISTINCT
        target_list_product
      , sum(available_qty) total_in_stock
      , current_date_in_stock
      FROM
        (view_running_inventory vri
      LEFT JOIN (
         SELECT DISTINCT
           ndc
         , target_list_product
         FROM
           iqvia_target_primary
      )  iq ON (vri.product_id = iq.ndc))
      WHERE ((current_date_in_stock >= date_add('year', -1, current_date)) AND (target_list_product IS NOT NULL) AND (donor_id <> '702') AND (NOT (target_list_product IN ('None', 'null'))))
      GROUP BY 1, 3
      ORDER BY 1 ASC
   ) 
   WHERE (total_in_stock > 0)
   GROUP BY 1
) 
SELECT DISTINCT
  drd2.target_list_product
, min_ship_date
, max_ship_date
, (max(CAST(total_units_distributed AS double)) / max(CAST(number_of_days_in_stock AS double))) daily_throughput
, round(((max(CAST(total_units_distributed AS double)) / max(CAST(number_of_days_in_stock AS double))) * 261), 0) annual_projection
, (round(((max(CAST(total_units_distributed AS double)) / max(CAST(number_of_days_in_stock AS double))) * 261), 0) * 1.2E0) annual_projection_with_buffer
, begin_date_stock_count
, end_date_stock_count
, total_units_distributed
, total_units_received
, (CASE WHEN (begin_date_stock_count IS NULL) THEN total_units_received ELSE (begin_date_stock_count + total_units_received) END) total_stock
, number_of_days_in_stock
FROM
  (drd2
LEFT JOIN vri2 ON (drd2.target_list_product = vri2.target_list_product))
WHERE ((number_of_days_in_stock > 50) AND ((total_units_received > 500) OR (begin_date_stock_count > 500)))
GROUP BY 1, 2, 3, 7, 8, 9, 10, 12
