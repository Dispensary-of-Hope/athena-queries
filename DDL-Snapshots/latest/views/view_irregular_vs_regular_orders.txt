WITH
  Regular_orders AS (
   SELECT
     dispensary_name
   , drug_name
   , CAST("avg"(ship_qty) AS int) avg_order_qty
   FROM
     "view_irregular_orders"
   WHERE (ship_date < (current_date - INTERVAL  '1' DAY))
   GROUP BY dispensary_name, drug_name
) 
SELECT
  (CASE WHEN (vio.ship_qty >= (2 * ro.avg_order_qty)) THEN 'Irregular Order' ELSE 'Regular Order' END) Order_type
, vio.dispensary_name
, vio.drug_name
, ro.avg_order_qty
, vio.ship_qty order_qty
, vio.order_date
, vio.ship_date
FROM
  ("view_irregular_orders" vio
INNER JOIN Regular_orders ro ON ((vio.dispensary_name = ro.dispensary_name) AND (vio.drug_name = ro.drug_name)))
GROUP BY 1, 2, 3, 4, 5, 6, 7
ORDER BY 2 ASC
