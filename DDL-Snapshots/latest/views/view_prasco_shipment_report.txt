SELECT
  ps.ponumber po_number
, ps.shipmentnumber shipment_id
, date_parse(ps.orderdate, '%Y-%m-%dT%H:%i:%s') order_date
, date_parse(ps.orderdate, '%Y-%m-%dT%H:%i:%s') ship_date
, ps.shiptoname dispensary_name
, ps.carrier
, ps.trackingnumber tracking_number
, replace(ps.productcode, '-', '') ndc
, pn.productnamelong
, TRY_CAST(ps.quantity AS BIGINT) qty
, ps.lotnumber lot
, date_parse(ps.lotexp, '%m/%d/%Y') expiration_date
, ps.shiptoaddress1 shippingaddress
, ps.shiptoaddress2 shippingaddress2
, ps.shiptocity shippingcity
, ps.shiptostate shippingstate
, ps.shiptozip shippingpostalcode
FROM
  (prasco_shipment ps
LEFT JOIN (
   SELECT
     ndc
   , arbitrary(productnamelong) productnamelong
   FROM
     view_prasco_inventory_report
   GROUP BY ndc
)  pn ON (replace(ps.productcode, '-', '') = pn.ndc))
