SELECT
  ocid.company_id__c
, ocid.site_id
, ocid.order_id
, ship_date
, order_date
, type
, status
, ocid.product_id
, drug_name
, total_units_distributed
, (CASE WHEN (donor_info.donor_id = '702') THEN 'Yes' ELSE 'No' END) Direct_Purchase
, "split_part"(ocid.lot_id, ' ', 3) lot_id
, (CASE WHEN regexp_like(split_part(ocid.lot_id, ' ', 2), '^[0-9]{1,2}/0/[0-9]{4}$') THEN CONCAT(substr(split_part(ocid.lot_id, ' ', 2), 1, (strpos(split_part(ocid.lot_id, ' ', 2), '/') - 1)), '/', '30/', split_part(split_part(ocid.lot_id, ' ', 2), '/', 3)) WHEN (ocid.lot_id = 'Mfg: 1/31/0223 3109396') THEN '1/31/2023' WHEN regexp_like(split_part(ocid.lot_id, ' ', 2), '^[0-9]{1,2}/[0-9]{1,2}/[0-9]{4}$') THEN split_part(ocid.lot_id, ' ', 2) WHEN regexp_like(split_part(ocid.lot_id, ' ', 2), '^[0-9]{4}-[0-9]{2}-[0-9]{2}$') THEN date_format(CAST(split_part(ocid.lot_id, ' ', 2) AS date), '%m/%d/%Y') ELSE null END) exp_date
, (CASE WHEN (donor_info.donor_id = '702') THEN mdp.seller_name ELSE donor_name END) donor_name
, (CASE WHEN (donor_info.donor_id = '702') THEN mdp.seller_address ELSE donor_info.donor_shipping_address END) donor_shipping_address
, 'Dispensary of Hope - Main Warehouse' seller_name
, '2700 Brick Church Pike, Nashville, TN 37207 USA' seller_address
, site_name buyer_name
, "concat"(shippingstreet, ', ', shippingcity, ', ', shippingstate, ', ', shippingpostalcode, ' USA') buyer_address
, (CASE WHEN (donor_info.donor_id = '702') THEN gs1_mfg_name ELSE manufacturer END) manufacturer
, gs1_mfg_address
, ohft.receive_date
FROM
  ((((orders_with_companyid ocid
LEFT JOIN (
   SELECT *
   FROM
     site_info
)  si ON (ocid.site_id = si.site_id))
LEFT JOIN (
   SELECT
     product_id
   , min(receive_date) receive_date
   , lot_id
   FROM
     orders_historical_finale_timestamp
   WHERE (receive_date IS NOT NULL)
   GROUP BY 1, 3
)  ohft ON ((ocid.product_id = ohft.product_id) AND (ocid.lot_id = ohft.lot_id)))
LEFT JOIN (
   SELECT DISTINCT
     donor_name
   , product_id
   , donor_id
   , donor_shipping_address
   FROM
     ((
      SELECT DISTINCT
        "max"(name) donor_name
      , product_id
      , "max"(donor_id) donor_id
      FROM
        (donor_id_finale dif
      LEFT JOIN (
         SELECT DISTINCT
           name
         , donor_id__c
         FROM
           sf_account
      )  sf ON (dif.donor_id = sf.donor_id__c))
      GROUP BY 2
   )  sf1
   LEFT JOIN (
      SELECT
        donor_id__c
      , concat(replace(shippingstreet, '"', ''), ', ', shippingcity, ', ', shippingstate, ' ', shippingpostalcode, ', ', shippingcountry) donor_shipping_address
      FROM
        sf_account
   )  sf2 ON (sf1.donor_id = sf2.donor_id__c))
)  donor_info ON (ocid.product_id = donor_info.product_id))
LEFT JOIN (
   SELECT DISTINCT
     product_id
   , gs1_mfg_name
   , replace(gs1_mfg_address, '"', '') gs1_mfg_address
   , seller_name
   , replace(seller_address, '"', '') seller_address
   FROM
     mckesson_dscsa_drug_purchasing
)  mdp ON (ocid.product_id = mdp.product_id))
WHERE (ocid.company_id__c <> '')
