(
   SELECT DISTINCT
     st.site_name
   , ehope_type__c
   , account_manager
   , st.weighted_score
   FROM
     ("database-prod"."site_stratification_clinics" st
   INNER JOIN (
      SELECT
        site_name
      , "max"(weighted_score) weighted_score
      FROM
        site_stratification_clinics
      GROUP BY site_name
   )  st2 ON ((st.site_name = st2.site_name) AND (st.weighted_score = st2.weighted_score)))
)
