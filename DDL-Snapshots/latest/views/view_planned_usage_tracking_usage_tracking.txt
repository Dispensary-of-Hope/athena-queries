SELECT
  dispensary_name
, productnamelong drug_name
, product_id
, lot_id
, "split_part"(lot_id, ' ', 3) donor_lot
, ship_date
, qty units_total
, (CASE WHEN (donor_id = '2192') THEN 'Planned' ELSE 'Excess' END) source_type
, 'Shipped' status
FROM
  donor_report_1
WHERE ((ship_date IS NOT NULL) AND (site_id IN ('214', '10233', '10276')))
UNION ALL (
   SELECT
     sf.name dispensary_name
   , drug_name
   , fco.product_id
   , null lot_id
   , null donor_lot
   , TRY_CAST(TRY(date_format(date_parse(order_date, '%m/%d/%Y'), '%Y-%m-%d %T')) AS timestamp) ship_date
   , TRY_CAST(total_units_distributed AS bigint) units_total
   , (CASE WHEN (donor_id = '2192') THEN 'Planned' ELSE 'Excess' END) source_type
   , 'Awaiting Fulfillment' status
   FROM
     ((finale_committed_orders fco
   LEFT JOIN (
      SELECT DISTINCT
        name
      , id__c
      FROM
        sf_account
   )  sf ON (fco.site_id = sf.id__c))
   LEFT JOIN (
      SELECT DISTINCT
        product_id
      , donor_id
      FROM
        donor_id_finale
   )  dif ON (fco.product_id = dif.product_id))
   WHERE (site_id IN ('10233', '10276', '214'))
)
