SELECT DISTINCT
  epro.ndc11
, epro.productnamelong
, epro.packageprice
, dcc.main_drug_class
, epro.drug_class
, epro.therapeutic_class
, epro.pharm_category
, epro.packagesize
, epro.unitname
, epro.packageid
, epro.companyname
, epro.companynameshort
, epro.statusname
, epro.unit_of_measure
, epro.mechanism_of_action
FROM
  (elsevier_2_aggregate_drugclass epro
LEFT JOIN drug_class_convert dcc ON (epro.drug_class = dcc.aggregated_drug_class))
