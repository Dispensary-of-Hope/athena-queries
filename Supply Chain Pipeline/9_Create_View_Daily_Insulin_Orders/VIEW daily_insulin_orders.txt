CREATE OR REPLACE VIEW "view_daily_insulin_orders" AS
(
SELECT distinct oid.product_id
, oid.status
, oid.drug_name
, oid.manufacturer
, oid.insulin
, oid.order_date
, oid.order_id
, oid.total_units_distributed
, sf.name dispensary_name
, count (distinct order_date) num_insulin_orders
FROM "database-prod"."orders_with_companyid" oid
LEFT JOIN sf_account sf ON sf.company_id__c = oid.company_id__c
WHERE oid.insulin = 'True' AND sf.name IS NOT NULL
Group by 1,2,3,4,5,6,7,8,9
Order BY oid.order_date DESC
);
