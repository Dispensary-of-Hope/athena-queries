CREATE EXTERNAL TABLE donor_id_iqvia ( 
`donor_id` string, 
`iqvia_donor` string,
`donor_name` string )

 ROW FORMAT SERDE 'org.apache.hadoop.hive.serde2.OpenCSVSerde'
 WITH SERDEPROPERTIES (
  'separatorChar' = ',',
  'quoteChar' = '\"',
  'escapeChar' = '\n'
 ) LOCATION 's3://iqviatables/iqvia_donors/'
 TBLPROPERTIES ('has_encrypted_data'='false',
               'skip.header.line.count' = '1');
