-- part of the IQVIA target primary pipeline
-- merges target list NDCs from iqvia data and NDCs not found in elsevier into one table
-- data for IQVIA target list NDCs and NDCs not found in elsevier is stored in S3
-- this is manually updated every quarter

CREATE EXTERNAL TABLE iqvia_target_primary ( target_list_product string, `priority` string, `primary` string, `combined_molecule` string, `strength` string, `product_sum` string, `rx_status` string, drug_class string,  `ndc` string, packagesize string )
ROW FORMAT SERDE 'org.apache.hadoop.hive.serde2.OpenCSVSerde'
WITH SERDEPROPERTIES (
  'separatorChar' = ',',
  'quoteChar' = '"',
  'escapeChar' = '\n'
)
LOCATION 's3://iqviatables/target_primary_lists/'
TBLPROPERTIES ('has_encrypted_data'='false', 'skip.header.line.count' = '1'
)
