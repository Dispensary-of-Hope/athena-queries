CREATE OR REPLACE VIEW "view_site_strat_weighted_score" AS (SELECT Distinct st.site_name, ehope_type__c, account_manager,    st.weighted_score,CASE    WHEN st.weighted_score < 40 THEN 'Low'    WHEN st.weighted_score < 61 AND st.weighted_score > 39 THEN 'Medium'    ELSE 'High'END AS high_medium_low_riskFROM "database-prod"."site_stratification" stINNER JOIN    (SELECT site_name, Max(weighted_score) weighted_score    FROM site_stratification Group BY site_name) st2        ON st.site_name = st2.site_name AND st.weighted_score = st2.weighted_score)
