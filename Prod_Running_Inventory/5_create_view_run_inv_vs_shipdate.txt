CREATE OR REPLACE VIEW "view_run_inv_vs_shipdate" AS 
( 
SELECT ldis.productnamelong,
ldis.product_id, ldis.current_date_in_stock last_day_in_stock,
Lsd.last_ship_date,
SUM(vri.total_in_stock) total_last_day_in_stock 
FROM


-- selecting max date in stock, product name, and product_id 
(SELECT DISTINCT MAX(current_date_in_stock) current_date_in_stock,
productnamelong,
product_id 
FROM view_running_inventory 
WHERE total_in_stock > 0 
GROUP BY 2,3) ldis 


--joining to get total in stock amount for the max date in stock 
-- left join to pull all ndcs previously in stock  
LEFT JOIN (SELECT SUM(total_in_stock) total_in_stock,
Current_date_in_stock,
Product_id,
productnamelong 
FROM view_running_inventory 
GROUP BY 2,3,4) vri 
ON ldis.product_id = vri.product_id AND ldis.current_date_in_stock = vri.current_date_in_stock 


--joining to get last ship date per product_id currently or previously in stock 
left join  
-- left join to pull all ndcs previously in stock  
(SELECT DISTINCT product_id,
MAX(ship_date) last_ship_date
FROM orders_with_companyid ocid 


--filtering out NDCs going to destruction 
WHERE ocid.site_id NOT IN ('10277', '194')  
GROUP BY 1)lsd 
ON ldis.product_id = lsd.product_id 
GROUP BY 1,2,3,4 
)
